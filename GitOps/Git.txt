ls - list all files
mkdir -  make directory
cd - change directory
git init - initialized git repo
touch . [file name] & file extension- creates new file
git sttaus - shows the untracked files
git add . - untracked files are staged
cat [file name] - shows what is in the files
git restore --staged [file name] - untracked files are removed from the staging area
git log - shows the commits history
rm -rf [file name] - file will be deleted or removed
